def realNormalMode {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
    (Ω α β : ℝ) (u v : E) (t : ℝ) : E :=
  α • (Real.sin (Ω * t) • u + Real.cos (Ω * t) • v) +
    β • (Real.cos (Ω * t) • u - Real.sin (Ω * t) • v)
