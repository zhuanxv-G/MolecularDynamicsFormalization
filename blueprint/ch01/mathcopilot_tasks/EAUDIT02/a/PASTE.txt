def schrodingerEquation (h : planckConstant) (μ : quantumMass)

  ContDiff ℝ 2 (Function.uncurry Φ) ∧

    -(h.val : ℂ)^2 * ∑ i : Fin 39,
      secondPartial (Φ t) q i / (2 * (μ ⟨i.val / 3, by omega⟩).val : ℂ) +
      (U q : ℂ) * Φ t q
