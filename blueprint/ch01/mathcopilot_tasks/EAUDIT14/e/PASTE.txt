def IsFirstIntegralOn (f : E → E) (Q : Set E) (J : E → ℝ) : Prop :=
  ∀ (a b : ℝ) (γ : ℝ → E),
    (∀ t ∈ Ioo a b, γ t ∈ Q) →
    (∀ t ∈ Ioo a b, HasDerivAt γ (f (γ t)) t) →
    ∀ s ∈ Ioo a b, ∀ t ∈ Ioo a b, J (γ s) = J (γ t)
