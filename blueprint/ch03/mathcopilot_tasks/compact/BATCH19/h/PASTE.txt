def partitionedRKRelation {n s : ℕ} (f : Z n → Z n)
    (Aq Ap : Matrix (Fin s) (Fin s) ℝ) (bq bp : Fin s → ℝ) (h : ℝ)
    (z w : Z n) (F : Fin s → Z n) : Prop :=
  (∀ i, F i=f (z.1+h • ∑ j, Aq i j • (F j).1,z.2+h • ∑ j, Ap i j • (F j).2)) ∧
    w=(z.1+h • ∑ i, bq i • (F i).1,z.2+h • ∑ i, bp i • (F i).2)
