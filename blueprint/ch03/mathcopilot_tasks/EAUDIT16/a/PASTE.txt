```lean
def complexEigenvalue {ι : Type*} [Fintype ι] [DecidableEq ι] (A : Matrix ι ι ℝ) (ζ : ℂ) : Prop :=
  ∃ v : ι → ℂ, v ≠ 0 ∧ (A.map (fun x => (x : ℂ))).mulVec v = ζ • v
```
