  (∑ i, x.2 i^2/(2*m i))+U x.1
def gibbsWeight (m : V n) (U : V n → ℝ) (β : ℝ) (x : Phase n) : ℝ :=
  Real.exp (-β*mechanicalHamiltonian m U x)

  β*∑ i, x.2 i/m i*gradient U x.1 i

  ouForward m γ β⁻¹ (fun p ↦ ρ (x.1,p)) x.2
